// KÜSIMUSED
const economicQuestions = [

const socialQuestions = [

    "Kliimamuutuste leevendamine ja rohepööre peavad olema riiklik prioriteet, isegi kui see tõstab maksukoormust.",

// +2 = täiesti nõus, -2 = üldse ei nõustu
const answers = [
    {label:"Täiesti nõus", value:2},
    {label:"Pigem nõus", value:1},
    {label:"Neutraalne", value:0},
    {label:"Pigem ei nõustu", value:-1},
    {label:"Üldse ei nõustu", value:-2}
];

// Majanduslik telg:
// - = vasakpoolne, + = parempoolne
// Positiivne vastus vasakpoolsele väitele vähendab X-i.
const economicDirection = [
    -1, -1, 1, -1, -1, -1, -1, -1, 1
];

// Sotsiaalkultuuriline telg:
// + = liberaalne/progressiivne
// - = konservatiivne
const socialDirection = [
    1, 1, -1, -1, 1, 1, -1, -1
];

// Erakondade koordinaadid
const parties = [
    {name:"Sotsiaaldemokraatlik Erakond (SDE)", x:-10, y:10},
    {name:"Keskerakond", x:-7.5, y:-4.5},
    {name:"Reformierakond", x:10, y:5},
    {name:"Eesti 200", x:8, y:9},
    {name:"Isamaa", x:8.5, y:-7},
    {name:"EKRE", x:7.5, y:-10}
];

const answerLabels = answers.map(a => a.label);

function renderQuestions() {
    [
        {id:"economic", questions:economicQuestions, prefix:"e"},
        {id:"social", questions:socialQuestions, prefix:"s"}
    ].forEach(group => {
        const container = document.getElementById(group.id);

        group.questions.forEach((question, i) => {
            const name = group.prefix + i;

            const div = document.createElement("div");
            div.className = "question";

            div.innerHTML = `
                <p>${i+1}. ${question}</p>
                <div class="answers">
                ${answers.map(a => `
                    <label>
                        <input type="radio" name="${name}" value="${a.value}" required>
                        ${a.label}
                    </label>
                `).join("")}
                </div>
            `;

            container.appendChild(div);
        });
    });
}

renderQuestions();

// SALVESTAMINE
function saveAnswers() {
    const data = {};

    document.querySelectorAll("#quiz input:checked").forEach(input => {
        data[input.name] = input.value;
    });

    localStorage.setItem("politicalQuizAnswers", JSON.stringify(data));
}

function loadAnswers() {
    const saved = JSON.parse(
        localStorage.getItem("politicalQuizAnswers") || "{}"
    );

    Object.entries(saved).forEach(([name,value]) => {
        const input = document.querySelector(
            `input[name="${name}"][value="${value}"]`
        );
        if(input) input.checked = true;
    });
}

document.getElementById("quiz").addEventListener("change", saveAnswers);

loadAnswers();

// ARVUTAMINE
function calculate() {
    let economic = 0;
    let social = 0;

    economicQuestions.forEach((q,i) => {
        const value = Number(
            document.querySelector(`input[name="e${i}"]:checked`).value
        );
        economic += value * economicDirection[i];
    });

    socialQuestions.forEach((q,i) => {
        const value = Number(
            document.querySelector(`input[name="s${i}"]:checked`).value
        );
        social += value * socialDirection[i];
    });

    // Normaliseerimine vahemikku -10 kuni +10
    const x = Math.round(economic / (economicQuestions.length * 2) * 100) / 10;
    const y = Math.round(social / (socialQuestions.length * 2) * 100) / 10;

    return {x,y};
}

// LÄHIMA ERAKONNA LEIDMINE
function findClosest(x,y) {
    return parties.map(p => ({
        ...p,
        distance: Math.sqrt(
            Math.pow(x-p.x,2) + Math.pow(y-p.y,2)
        )
    })).sort((a,b) => a.distance-b.distance);
}

// TULEMUSED
document.getElementById("quiz").addEventListener("submit", function(e) {
    e.preventDefault();

    if(!this.reportValidity()) return;

    saveAnswers();

    const {x,y} = calculate();
    const sorted = findClosest(x,y);

    document.getElementById("results").style.display = "block";
    document.getElementById("closestParty").textContent =
        "Sinu vaadetele on kõige lähemal: " + sorted[0].name;

    document.getElementById("description").textContent =
        `Sinu poliitiline asukoht on (${x}; ${y}). ` +
        "Erakondade lähedus arvutatakse nende koordinaatide põhjal.";

    document.getElementById("xScore").textContent = x;
    document.getElementById("yScore").textContent = y;

    document.getElementById("partyList").innerHTML = sorted.map((p,i) => `
        <div class="party">
            <b>${i+1}. ${p.name}</b><br>
            <small>Koordinaadid: (${p.x}; ${p.y}) · Kaugus: ${p.distance.toFixed(2)}</small>
        </div>
    `).join("");

    drawChart(x,y);

    document.getElementById("results").scrollIntoView({
        behavior:"smooth"
    });
});

// KOORDINAATTELJESTIK
function drawChart(x,y) {
    const canvas = document.getElementById("chart");
    const ctx = canvas.getContext("2d");

    const w = canvas.width;
    const h = canvas.height;
    const padding = 65;

    const scale = (w - 2*padding) / 20;

    function px(v) {
        return padding + (v+10)*scale;
    }

    function py(v) {
        return h-padding - (v+10)*scale;
    }

    ctx.clearRect(0,0,w,h);

    // Taustavõrgustik
    ctx.strokeStyle = "#e2e8f0";
    ctx.lineWidth = 1;

    for(let i=-10;i<=10;i+=2) {
        ctx.beginPath();
        ctx.moveTo(px(i),py(-10));
        ctx.lineTo(px(i),py(10));
        ctx.stroke();

        ctx.beginPath();
        ctx.moveTo(px(-10),py(i));
        ctx.lineTo(px(10),py(i));
        ctx.stroke();
    }

    // Teljed
    ctx.strokeStyle = "#334155";
    ctx.lineWidth = 2;

    ctx.beginPath();
    ctx.moveTo(px(-10),py(0));
    ctx.lineTo(px(10),py(0));
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(px(0),py(-10));
    ctx.lineTo(px(0),py(10));
    ctx.stroke();

    ctx.fillStyle = "#334155";
    ctx.font = "13px Arial";
    ctx.textAlign = "center";
    ctx.fillText("Majanduslik telg",w/2,h-13);
    ctx.fillText("−10",px(-10),h-45);
    ctx.fillText("0",px(0),h-45);
    ctx.fillText("+10",px(10),h-45);

    ctx.save();
    ctx.translate(17,h/2);
    ctx.rotate(-Math.PI/2);
    ctx.fillText("Sotsiaalkultuuriline telg",0,0);
    ctx.restore();

    // Erakonnad
    parties.forEach(p => {
        ctx.fillStyle = "#64748b";
        ctx.beginPath();
        ctx.arc(px(p.x),py(p.y),6,0,Math.PI*2);
        ctx.fill();

        ctx.fillStyle = "#334155";
        ctx.font = "11px Arial";
        ctx.textAlign = "center";
        ctx.fillText(p.name.split(" ")[0],px(p.x),py(p.y)-12);
    });

    // Vastaja punkt
    ctx.fillStyle = "#2563eb";
    ctx.beginPath();
    ctx.arc(px(x),py(y),10,0,Math.PI*2);
    ctx.fill();

    ctx.strokeStyle = "white";
    ctx.lineWidth = 3;
    ctx.stroke();

    ctx.fillStyle = "#2563eb";
    ctx.font = "bold 13px Arial";
    ctx.fillText("Sina",px(x),py(y)-18);
}

// UUESTI ALUSTAMINE
function resetQuiz() {
    if(!confirm("Kas soovid kõik vastused kustutada?")) return;

    localStorage.removeItem("politicalQuizAnswers");
    document.getElementById("quiz").reset();
    document.getElementById("results").style.display = "none";
}
